# tools/dxvk/cross-win32-posix.txt — DXVK win32 cross file for THIS host.

# Mirrors cross-win64-posix.txt: the bare i686-w64-mingw32-* tools default to
# the win32-threads flavor, and DXVK does not build with win32 threads
# (std::cv_status has not been declared). Pin the -posix suffixed tools.

ar = 'i686-w64-mingw32-ar'
strip = 'i686-w64-mingw32-strip'
windres = 'i686-w64-mingw32-windres'

[properties]
needs_exe_wrapper = true

[host_machine]
system = 'windows'
cpu_family = 'x86'
cpu = 'x86'
endian = 'little'
